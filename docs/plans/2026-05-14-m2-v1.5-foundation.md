- [ ] **Step 6: RLS 정책**

`supabase/migrations/20260514000006_rls.sql`:
```sql
-- ai_chats
alter table ai_chats enable row level security;

create policy ai_chats_select on ai_chats for select using (
  case visibility
    when 'pair' then is_couple_member(couple_id)
    when 'private' then owner_user_id = auth.uid()
  end
);
create policy ai_chats_insert on ai_chats for insert with check (
  owner_user_id = auth.uid() and is_couple_member(couple_id)
);
create policy ai_chats_update on ai_chats for update using (
  owner_user_id = auth.uid()
);
create policy ai_chats_delete on ai_chats for delete using (
  owner_user_id = auth.uid()
);

-- ai_messages (chat의 visibility 상속)
alter table ai_messages enable row level security;

create policy ai_messages_select on ai_messages for select using (
  exists (
    select 1 from ai_chats c
    where c.id = chat_id
      and (
        (c.visibility = 'pair' and is_couple_member(c.couple_id))
        or (c.visibility = 'private' and c.owner_user_id = auth.uid())
      )
  )
);
create policy ai_messages_insert on ai_messages for insert with check (
  exists (
    select 1 from ai_chats c
    where c.id = chat_id and c.owner_user_id = auth.uid()
  )
);

-- diary_entries
alter table diary_entries enable row level security;

create policy diary_select on diary_entries for select using (
  case visibility
    when 'pair' then is_couple_member(couple_id)
    when 'private' then author_user_id = auth.uid()
  end
);
create policy diary_insert on diary_entries for insert with check (
  author_user_id = auth.uid() and is_couple_member(couple_id)
);
create policy diary_update on diary_entries for update using (
  author_user_id = auth.uid()
);
create policy diary_delete on diary_entries for delete using (
  author_user_id = auth.uid()
);

-- rag_chunks (인증 사용자 read · service_role만 write)
alter table rag_chunks enable row level security;
create policy rag_chunks_select on rag_chunks for select to authenticated using (true);

-- personal_context
alter table personal_context enable row level security;

create policy personal_context_select on personal_context for select using (
  is_couple_member(couple_id)
);
create policy personal_context_insert on personal_context for insert with check (
  is_couple_member(couple_id)
);
create policy personal_context_update on personal_context for update using (
  is_couple_member(couple_id)

// lib/ai/system-prompt.ts (continued)
import { P1_FRAGMENT } from "./persona-prompts/p1";
import { P2_FRAGMENT } from "./persona-prompts/p2";
import { P3_FRAGMENT } from "./persona-prompts/p3";
import { P4_FRAGMENT } from "./persona-prompts/p4";

type Persona = 'P1' | 'P2' | 'P3' | 'P4' | 'general';
type Visibility = 'pair' | 'private';

export type PersonalContext = {
  cycleStage?: string;
  cycleNumber?: number;
  currentMedications?: Record<string, unknown>;
  recentEmotionalState?: string;
  recentCoupleIssues?: string;
};

export function buildSystemPrompt(opts: {
  persona: Persona;
  visibility: Visibility;
  personalContext?: PersonalContext;
}): string {
  const parts = [SYSTEM_PROMPT_BASE];

  const fragments: Record<Persona, string> = {
    P1: P1_FRAGMENT,
    P2: P2_FRAGMENT,
    P3: P3_FRAGMENT,
    P4: P4_FRAGMENT,
    general: '',
  };
  if (fragments[opts.persona]) parts.push(fragments[opts.persona]);

  parts.push(`\n# 이 conversation visibility: ${opts.visibility}\n${
    opts.visibility === 'pair'
      ? 'pair: 부부 양쪽 모두에게 보임. 양쪽이 이 대화 참여할 수 있음.'
      : 'private: 본인만 보임. 부부 갈등·민감 의료정보 자유롭게 OK.'
  }`);

  if (opts.personalContext && Object.keys(opts.personalContext).length > 0) {
    const pc = opts.personalContext;
    parts.push(`\n# Personal context (per turn)
- 사이클: ${pc.cycleNumber ? `${pc.cycleNumber}차` : '미입력'}${pc.cycleStage ? `, ${pc.cycleStage}` : ''}
- 현 약: ${pc.currentMedications ? JSON.stringify(pc.currentMedications) : '미입력'}
- 최근 감정 상태: ${pc.recentEmotionalState ?? '미입력'}
- 최근 부부 이슈: ${pc.recentCoupleIssues ?? '미입력'}

이미 알고 있는 정보는 다시 묻지 마세요.`);
  }

  return parts.join('\n\n');
}

const STAT_NUMBER_PATTERN = /(\d{1,3}(\.\d+)?)\s*(%|퍼센트|확률)/g;
const SOURCE_PATTERN = /\(출처[:\s]/;

export type StatGuardResult = {
  hasStats: boolean;
  hasSource: boolean;
  numbers: string[];
  warning?: string;
};

export function checkStatisticsCitation(text: string): StatGuardResult {
  const numbers = Array.from(text.matchAll(STAT_NUMBER_PATTERN)).map(m => m[0]);
  const hasStats = numbers.length > 0;
  const hasSource = SOURCE_PATTERN.test(text);
  if (hasStats && !hasSource) {
    return {
      hasStats: true,
      hasSource: false,
      numbers,
      warning: '통계 수치는 있으나 출처 표기 없음 (K1 Guard-2 violation)',
    };
  }
  return { hasStats, hasSource, numbers };
}

// lib/ai/persona-router.ts
export type Persona = 'P1' | 'P2' | 'P3' | 'P4' | 'general';

export type UserMeta = {
  cycleCount?: number;
  yearsTrying?: number;
  maleFactor?: boolean;
};

const P2_KEYWORDS = [
  '그만', '마지막', '포기', '끝내', '못 견디', '못해',
  '죽고 싶', '죽을', '안 되겠',
];
const P3_KEYWORDS = ['정자', '내 탓', '남편 탓', '남자 구실', '내가 발목'];
const P4_KEYWORDS = ['1년', '2년', '난임센터 가야', '시술 진입', '자연시도'];

function hasKeyword(input: string, list: string[]): boolean {
  return list.some(kw => input.includes(kw));
}

export function classifyPersona(input: string, meta: UserMeta = {}): Persona {
  // 우선순위: P2 (위험군) > P3 > P4 > P1
  if ((meta.cycleCount ?? 0) >= 3) return 'P2';
  if (hasKeyword(input, P2_KEYWORDS)) return 'P2';

  if (meta.maleFactor || hasKeyword(input, P3_KEYWORDS)) return 'P3';

  if ((meta.cycleCount ?? 0) === 0 && (meta.yearsTrying ?? 0) >= 1) return 'P4';
  if (hasKeyword(input, P4_KEYWORDS) && (meta.cycleCount ?? 0) === 0) return 'P4';

  if ((meta.cycleCount ?? 0) === 0) return 'P1';

  return 'general';
}
```

  /번아웃/, /무너졌/, /내가\s*나한테\s*놀랐/,
];

export type MentalHealthGuardResult = {
  triggered: boolean;
  matched: string[];
};

export function detectMentalHealthEmergency(input: string): MentalHealthGuardResult {
  const matched: string[] = [];
  for (const pat of MENTAL_HEALTH_EMERGENCY_PATTERNS) {
    const m = input.match(pat);
    if (m) matched.push(m[0]);
  }
  return { triggered: matched.length > 0, matched };
}
